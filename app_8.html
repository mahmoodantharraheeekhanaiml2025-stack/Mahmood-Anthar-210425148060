import heapq
from itertools import permutations

INF = float('inf')

def tsp_brute_force(cost, n):
    cities = list(range(1, n))
    best_cost = INF
    best_path = None
    for perm in permutations(cities):
        path = [0] + list(perm) + [0]
        c = sum(cost[path[i]][path[i+1]] for i in range(n))
        if c < best_cost:
            best_cost = c
            best_path = path
    return best_path, best_cost

# 5-city cost matrix
cost = [
    [INF,  10,   8,   9,   7],
    [ 10, INF,  10,   5,   6],
    [  8,  10, INF,   8,   9],
    [  9,   5,   8, INF,   6],
    [  7,   6,   9,   6, INF]
]
n = 5
cities = ['A', 'B', 'C', 'D', 'E']

best_path, best_cost = tsp_brute_force(cost, n)
print('Optimal Tour:', " -> ".join(cities[i] for i in best_path))
print('Minimum Cost:', best_cost)
